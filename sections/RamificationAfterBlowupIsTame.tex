\section[The Cyclic Exceptional--Divisor Calculation and Final Assembly]
{The Cyclic Exceptional--Divisor Calculation and\newline Final Assembly}

We are now ready to prove the main result. The previous sections reduce the
problem to three cases, according to whether the cyclic group has order prime
to $p$, order $p$, or order $p^r$. The following theorem proves these three
cases, which together imply \Cref{theorem:torsors_after_blowup_is_tame}.

\begin{theorem}[The three cyclic cases]
\label{theorem:cyclic_cases_after_blowup}
\label{theorem:ramification_after_blowup_P1}
\label{cor:ramification_after_blowup_is_tame}
\label{theorem:ramification_after_blowup_is_tame_for_p_power}
\textcolor{red}{Assume that $k$ is algebraically closed.}
Let $\cP\to U=C\setminus\mdls$ be a $G$-torsor with $G$ cyclic, and let
$\ndls=dP\subset\mdls$ be a single-point submodulus.
\begin{enumerate}
    \item If $G=\Z/e\Z$ with $(e,p)=1$, then $\cP^{[d]}$ is tamely
    ramified at $\eta_\ndls$.
    \item If $G=\Z/p\Z$ and the ramification of $\cP$ at $P$ is bounded
    by $d$, then $\cP^{[d]}$ is unramified at $\eta_\ndls$.
    \item If $G=\Z/p^r\Z$, $r\geq2$, and the ramification of $\cP$ at
    $P$ is bounded by $d$, then $\cP^{[d]}$ is unramified at
    $\eta_\ndls$.
\end{enumerate}
\end{theorem}

\begin{proof}
Since $k$ is algebraically closed, $P$ is $k$-rational. Choose a uniformizer
$x$ at $P$. It identifies the completed local ring at $P$ with $k[[x]]$
and its fraction field with $k((x))$. Let $\cP_x$ be the restriction of
$\cP$ to the punctured formal disc $\Spec k((x))$. By
\Cref{cor:abelian_equivilance_fundamental_torsors}, the isomorphism class of
$\cP_x$ corresponds to a unique continuous homomorphism
\[
    \rho:\Gal(k((x))^{\mathrm{sep}}/k((x)))\longrightarrow G.
\]
Put $H:=\operatorname{im}(\rho)$. Under the construction of
\Cref{theorem:equivalence_fundamental_covers}, the surjection onto $H$
determines a connected $H$-torsor $\cQ$ over $\Spec k((x))$, and, for the
inclusion $\iota:H\hookrightarrow G$, one has
$\cP_x\cong\iota_*\cQ$.
If $H$ is trivial, then $\cP_x$ is trivial. Formal--local invariance
(\Cref{lemma:formal_local_invariance}) then shows that $\cP^{[d]}$ is
unramified at $\eta_\ndls$, which proves all three conclusions in this case.

Assume $H\neq0$. In parts~(2) and~(3), the assumed ramification bound passes
from $\cP_x$ to $\cQ$ by
\Cref{lemma:ramification_of_induced_torsor}(3), because $k[[x]]$ is complete
and its residue field $k$ is perfect.

The realization lemma (\Cref{lemma:realization_over_Gm}) gives an explicit
connected $H$-torsor $\cQ'$ on $\bG_m$ whose restriction to
$\Spec k((x))$ is $\cQ$. Hence the induced $G$-torsor $\iota_*\cQ'$ has the
same punctured-formal-disc restriction as $\cP$. The reduction to $\bP^1$
(\Cref{cor:reduction_to_P1}) therefore identifies the relevant ramification
of $\cP^{[d]}$ with that of $(\iota_*\cQ')^{[d]}$. The compatibility of
symmetric powers with induced torsors gives
\[
    (\iota_*\cQ')^{[d]}
       \cong\iota_*(\cQ'^{[d]})
\]
by \Cref{lemma:symmetric_power_induced_torsor}. Parts~(1) and~(2) of
\Cref{lemma:ramification_of_induced_torsor} then show that tameness or
unramifiedness of $\cQ'^{[d]}$ is equivalent to the same property for the
induced torsor. Thus it is enough to treat the connected explicit model
\[
    C=\bP^1_k,\qquad U=\bG_m,\qquad \ndls=d\cdot0.
\]
For the rest of the calculation, we rename the subgroup $H$ as $G$ and the
realized torsor $\cQ'$ as $\cP$.
In Case~(1), write $n:=|G|$ for the order of this renamed subgroup. Since it
is a subgroup of the original group $\Z/e\Z$, one has $n\mid e$, and hence
$(n,p)=1$.

Write $x_1,\dots,x_d$ for the coordinates on $\bG_m^d$ and
$e_1,\dots,e_d$ for their elementary symmetric polynomials. By
\Cref{lemma:formal_local_invariance}, the completed local ring and field at
the generic point $\eta=\eta_{d\cdot0}$ of the exceptional divisor are
\begin{equation}\label{eq:complete_field_at_generic_point}
    \widehat\Oo_\eta=\kappa[[e_d]],\qquad
    \widehat K_\eta=\kappa((e_d)),\qquad
    \kappa=k(e_1/e_d,\dots,e_{d-1}/e_d).
\end{equation}
Each of the three calculations follows the same plan. We write the local
torsor in a standard form, combine the $d$ equations using the relevant group
law, and check the resulting coefficient at the exceptional valuation. The
calculation has the following shape:
\[
\begin{array}{c|c|c|c}
 &\Z/n\Z&\Z/p\Z&\Z/p^r\Z\\ \hline
\text{local equation}&T^n=x^a&X^p-X=f&\wp\vec X=\vec f\\
\text{$d$-fold coordinate}&\prod_iT_i&\sum_iX_i&\bigoplus_i\vec X_i\\
\text{coefficient after descent}&e_d^a&\sum_i f(x_i)&\bigoplus_i\vec f(x_i)\\
\text{at the exceptional valuation}&v(e_d^a)=a&\text{regular}&
    \text{componentwise regular}\\
\text{conclusion}&\text{tame}&\text{unramified}&\text{unramified}.
\end{array}
\]

\medskip
\noindent\textbf{Case (1): Kummer.}
Put
\[
    R:=k[x,x^{-1}],\qquad U=\bG_m=\Spec R.
\]
By \Cref{lemma:realization_over_Gm}(1), the connected local torsor $\cQ$ is
represented by
\[
    T^n=x^a
\]
for some integer $a$ with $(a,n)=1$, and the realized torsor
$\cP=\cQ'\to U$ is defined by the same equation. Equivalently, because
$k$ is algebraically closed, we may choose an identification $\mu_n\cong G$,
and then $\cP$ fits into the Cartesian square
\[
\begin{tikzcd}
    \cP \arrow[r] \arrow[d]
        & \bG_m \arrow[d,"{[n]}"] \\
    U \arrow[r,"{x^a}"']
        & \bG_m,
\end{tikzcd}
\]
where $[n](T)=T^n$ and the lower horizontal arrow is the morphism
corresponding to the section $x^a\in\bG_m(U)=R^\times$. Thus $\cP$ is the
pullback of $[n]$ along the section $x^a$. Put
\[
    S:=R[T]/(T^n-x^a).
\]
Thus $\cP\cong\Spec S$ as a $G$-torsor over $U$. After pulling it
back along the $d$ projections $U^d\to U$, the product torsor has
coordinate algebra $S^{\otimes_k d}$ over $R^{\otimes_k d}$. Contracting the
product action $G^d$ along $G^d\to G$ means taking invariants under
$\KG=\ker(G^d\to G)$. Write $T_i$ for the pullback of the coordinate $T$
along the $i$-th projection from the product torsor. By
\Cref{prop:parallel_cyclic_product_torsor}, the contracted product of the
$d$ torsors
\[
    T_i^n=x_i^a
\]
is obtained by multiplying their defining sections. It therefore has the
equation
\[
    Y^n=\prod_{i=1}^d x_i^a
       =(x_1\cdots x_d)^a
       =e_d^a.
\]
Here $Y$ is the coordinate on the resulting torsor. Under the isomorphism
of \Cref{prop:parallel_cyclic_product_torsor}, its pullback to the product
torsor is $\prod_{i=1}^d T_i$. Consequently,
\[
    (S^{\otimes_k d})^{\KG}
       \cong
    R^{\otimes_k d}[Y]/(Y^n-e_d^a).
\]
The group $S_d$ permutes the tensor factors and fixes both $Y$ and $e_d^a$.
The algebra on the right is free over $R^{\otimes_k d}$ with the
$S_d$-invariant basis $1,Y,\dots,Y^{n-1}$, so taking $S_d$-invariants acts
only on its coefficients. Since
\[
    (R^{\otimes_k d})^{S_d}
       =k[x_1^{\pm1},\dots,x_d^{\pm1}]^{S_d}
       =k[e_1,\dots,e_d,e_d^{-1}],
\]
\Cref{prop:symmetric_power_semidirect_quotient} identifies the descended torsor with
\[
    \cP^{[d]}\cong
    \Spec\frac{k[e_1,\dots,e_d,e_d^{-1}][Y]}{(Y^n-e_d^a)}.
\]
After base change to $\widehat K_\eta=\kappa((e_d))$, we obtain
\[
    \cP^{[d]}\big|_{\Spec\widehat K_\eta}
       \cong
    \Spec\frac{\widehat K_\eta[Y]}{(Y^n-e_d^a)}.
\]
Since $e_d$ is a uniformizer of $\widehat\Oo_\eta=\kappa[[e_d]]$, we have
\[
    v_\eta(e_d^a)=a.
\]
Together with $\gcd(a,n)=1$, \Cref{theorem:kummer_ramification} shows that
the Kummer extension defined by $Y^n=e_d^a$ is totally and tamely ramified.
Hence $\cP^{[d]}$ is tamely ramified at $\eta$, proving part~(1).

\medskip
\noindent\textbf{Case (2): Artin--Schreier.}
With $R=k[x,x^{-1}]$ and $U=\bG_m=\Spec R$ as in Case~(1),
\Cref{lemma:realization_over_Gm}(2) presents the realized torsor
$\cP\to U$ as
\[
    X^p-X=f(x),\qquad
    0\ne f\in x^{-1}k[x^{-1}],\qquad
    m:=\deg_{x^{-1}}f<d,\quad p\nmid m.
\]
Equivalently, $\cP$ is the pullback of the finite \'etale homomorphism
\[
    \wp:\bG_a\longrightarrow\bG_a,\qquad X\longmapsto X^p-X,
\]
whose kernel is $\F_p\cong G$, along the section
$f\in\bG_a(U)=R$. Put
\[
    S:=R[X]/(X^p-X-f(x)).
\]
Then $\cP\cong\Spec S$. As in Case~(1), the contracted product has coordinate
algebra $(S^{\otimes_k d})^{\KG}$, where
$\KG:=\ker(G^d\to G)$. Write $X_i$ for the pullback of $X$ from the $i$-th
factor. Since the group law on $\bG_a$ is addition,
\Cref{prop:parallel_cyclic_product_torsor} shows that the defining sections
$f(x_i)$ add, giving the equation
\[
    Y^p-Y=\alpha,
    \qquad
    \alpha:=\sum_{i=1}^d f(x_i).
\]
Here $Y$ is the coordinate on the contracted product, and its pullback to
the product torsor is $\sum_{i=1}^d X_i$. Consequently,
\[
    (S^{\otimes_k d})^{\KG}
       \cong
    R^{\otimes_k d}[Y]/(Y^p-Y-\alpha).
\]
As in Case~(1), $S_d$ fixes $Y$ and $\alpha$, and taking invariants acts
only on the coefficients. Thus
\Cref{prop:symmetric_power_semidirect_quotient} gives
\[
    \cP^{[d]}\cong
    \Spec\frac{k[e_1,\dots,e_d,e_d^{-1}][Y]}{(Y^p-Y-\alpha)}.
\]
Put $y_i=x_i^{-1}$. The polynomial $\alpha$ is symmetric in
$y_1,\dots,y_d$, and all its monomials have total degree at most $m<d$.
Hence
\Cref{lemma:symmetric_regularity} gives
\[
    \alpha\in\kappa\subset\widehat\Oo_\eta.
\]
Thus, although the individual $f(x_i)$ have poles at $x_i=0$, their
symmetric sum has no pole at the exceptional valuation.
Set $A:=\widehat\Oo_\eta$ and
\[
    B:=A[Y]/(Y^p-Y-\alpha).
\]
The defining polynomial is monic and has derivative $-1$, so $B/A$ is
finite \'etale. After base change to
$\widehat K_\eta=\kappa((e_d))$, we obtain
\[
    \cP^{[d]}\big|_{\Spec\widehat K_\eta}
       \cong
    \Spec\bigl(B\otimes_A\widehat K_\eta\bigr)
       =\Spec\frac{\widehat K_\eta[Y]}{(Y^p-Y-\alpha)}.
\]
Every field factor is therefore unramified over $\widehat K_\eta$. Hence
$\cP^{[d]}$ is unramified at $\eta$, proving part~(2).

\end{proof}

\begin{proof}[Proof of \Cref{theorem:torsors_after_blowup_is_tame}]
Part~(1) of \Cref{theorem:cyclic_cases_after_blowup} proves the conclusion
for cyclic groups of order prime to $p$, while parts~(2) and~(3) prove it
for cyclic $p$-groups. The result for every finite abelian group $G$ now
follows directly from \Cref{prop:reduction_to_cyclic_groups}.
\end{proof}
